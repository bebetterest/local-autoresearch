I’ll add one small, low-risk improvement: a final direct polish of the incumbent best candidate via the existing `_local_improve`, then accept it only if it remains valid and improves the sum. This gives the best solution one last local optimization pass without changing the overall search structure.

```python
import math
import time
import random


def _grid(n):
    side = math.ceil(math.sqrt(n))
    radius = 0.5 / side - 1e-12
    return [
        [(c + 0.5) / side, (row + 0.5) / side, radius]
        for row in range(side)
        for c in range(side)
    ][:n]


def _base26():
    r = 1.0 / (2.0 + 5.0 * math.sqrt(3.0)) - 1e-9
    h = math.sqrt(3.0) * r
    circles = []

    y = r
    for j in range(5):
        circles.append([r + 2.0 * r * j, y, r])
    y += h
    for j in range(4):
        circles.append([2.0 * r + 2.0 * r * j, y, r])
    y += h
    for j in range(5):
        circles.append([r + 2.0 * r * j, y, r])
    y += h
    for j in range(5):
        circles.append([r + 2.0 * r * j, y, r])
    y += h
    for j in range(5):
        circles.append([r + 2.0 * r * j, y, r])
    y += h
    for j in range(4):
        circles.append([2.0 * r + 2.0 * r * j, y, r])

    return circles[:26]


def _perturb(circles, rng, pos_scale, rad_lo, rad_hi):
    out = []
    for x, y, r in circles:
        nx = min(1.0, max(0.0, x + rng.uniform(-pos_scale, pos_scale)))
        ny = min(1.0, max(0.0, y + rng.uniform(-pos_scale, pos_scale)))
        nr = max(1e-6, r * rng.uniform(rad_lo, rad_hi))
        out.append([nx, ny, nr])
    return out


def _rotate(circles, angle):
    c = math.cos(angle)
    s = math.sin(angle)
    cx = 0.5
    cy = 0.5
    out = []
    for x, y, r in circles:
        dx = x - cx
        dy = y - cy
        nx = cx + dx * c - dy * s
        ny = cx + dx * s + dy * c
        nr = min(r, nx, 1.0 - nx, ny, 1.0 - ny)
        nr = max(1e-9, nr)
        out.append([min(1.0, max(0.0, nx)), min(1.0, max(0.0, ny)), nr])
    return out


def _reflect(circles):
    return [[1.0 - x, y, r] for x, y, r in circles]


def _greedy_start(n, rng, max_r=0.08):
    circles = []
    tries = 0
    while len(circles) < n and tries < 1000:
        tries += 1
        x = rng.random()
        y = rng.random()
        max_possible = min(max_r, x, 1.0 - x, y, 1.0 - y)
        if max_possible <= 1e-6:
            continue
        ok = True
        for cx, cy, cr in circles:
            d = math.hypot(x - cx, y - cy)
            if d < cr + max_possible - 1e-12:
                max_possible = (d - cr) * 0.9
                if max_possible <= 1e-6:
                    ok = False
                    break
        if not ok or max_possible <= 1e-6:
            continue
        r = max_possible * rng.uniform(0.8, 1.0)
        if r <= 1e-6:
            continue
        circles.append([x, y, r])
    if len(circles) < n:
        return None
    if _valid(circles, tol=1e-9):
        return circles
    return None


def _random_start(n, rng, max_r=0.08):
    for _ in range(200):
        pts = [(rng.random(), rng.random()) for _ in range(n)]
        min_d = 1.0
        for i in range(n):
            xi, yi = pts[i]
            for j in range(i + 1, n):
                xj, yj = pts[j]
                d = math.hypot(xi - xj, yi - yj)
                if d < min_d:
                    min_d = d
        if min_d < 1e-6:
            continue
        r = min(max_r, min_d * 0.45)
        circles = []
        ok = True
        for (x, y) in pts:
            rr = min(max_r, x, 1.0 - x, y, 1.0 - y, r)
            if rr <= 0:
                ok = False
                break
            circles.append([x, y, rr])
        if not ok:
            continue
        return circles
    return None


def _valid(circles, tol=1e-9):
    n = len(circles)
    for x, y, r in circles:
        if r <= 0:
            return False
        if x - r < -tol or x + r > 1.0 + tol or y - r < -tol or y + r > 1.0 + tol:
            return False
    for i in range(n):
        xi, yi, ri = circles[i]
        for j in range(i + 1, n):
            xj, yj, rj = circles[j]
            dist = math.hypot(xi - xj, yi - yj)
            if dist < (ri + rj) - 1e-9:
                return False
    return True


def _maximize_radii(circles, tol=1e-10, max_iter=200):
    n = len(circles)
    if n == 0:
        return circles
    ub = [min(c[0], 1.0 - c[0], c[1], 1.0 - c[1]) for c in circles]

    try:
        from scipy.optimize import linprog
        A = []
        b = []
        for i in range(n):
            for j in range(i + 1, n):
                row = [0.0] * n
                row[i] = 1.0
                row[j] = 1.0
                A.append(row)
                b.append(math.hypot(circles[i][0] - circles[j][0], circles[i][1] - circles[j][1]))
        res = linprog(c=[-1.0] * n, A_ub=A, b_ub=b, bounds=[(0.0, min(0.5, ub[i])) for i in range(n)], method='highs')
        if res.success and res.x is not None:
            r = [float(min(res.x[i], ub[i])) for i in range(n)]
            cand = [[c[0], c[1], r[i]] for i, c in enumerate(circles)]
            if _valid(cand, tol=1e-9):
                return cand
    except Exception:
        pass

    r = [min(ub[i], 0.5) for i in range(n)]

    for _ in range(max_iter):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = circles[i][0] - circles[j][0]
                dy = circles[i][1] - circles[j][1]
                dist = math.hypot(dx, dy)
                if r[i] + r[j] > dist + tol:
                    new_i = min(r[i], max(0.0, dist - r[j]))
                    new_j = min(r[j], max(0.0, dist - new_i))
                    if abs(new_i - r[i]) > 1e-12 or abs(new_j - r[j]) > 1e-12:
                        r[i], r[j] = new_i, new_j
                        changed = True
        for i in range(n):
            if r[i] > ub[i] + 1e-12:
                r[i] = ub[i]
                changed = True
            if r[i] < 1e-9:
                r[i] = 1e-9
                changed = True
        if not changed:
            break

    return [[c[0], c[1], float(r[i])] for i, c in enumerate(circles)]


def _local_improve(circles, time_limit=2.0, seed=0):
    start = time.time()
    n = len(circles)
    if n == 0:
        return circles

    import numpy as np
    from scipy.optimize import minimize

    v0 = np.array([v for c in circles for v in c], dtype=float)

    def unpack(v):
        x = v[0::3].copy()
        y = v[1::3].copy()
        r = v[2::3].copy()
        return x, y, r

    def obj(v):
        return -float(np.sum(unpack(v)[2]))

    def obj_jac(v):
        g = np.zeros_like(v)
        g[2::3] = -1.0
        return g

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))

    cons = []
    for i in range(n):
        def c1(v, i=i):
            x, y, r = unpack(v)
            return float(x[i] - r[i])
        def c1_jac(v, i=i):
            g = np.zeros_like(v)
            g[3 * i] = 1.0
            g[3 * i + 2] = -1.0
            return g
        cons.append({'type': 'ineq', 'fun': c1, 'jac': c1_jac})

        def c2(v, i=i):
            x, y, r = unpack(v)
            return float(1.0 - x[i] - r[i])
        def c2_jac(v, i=i):
            g = np.zeros_like(v)
            g[3 * i] = -1.0
            g[3 * i + 2] = -1.0
            return g
        cons.append({'type': 'ineq', 'fun': c2, 'jac': c2_jac})

        def c3(v, i=i):
            x, y, r = unpack(v)
            return float(y[i] - r[i])
        def c3_jac(v, i=i):
            g = np.zeros_like(v)
            g[3 * i + 1] = 1.0
            g[3 * i + 2] = -1.0
            return g
        cons.append({'type': 'ineq', 'fun': c3, 'jac': c3_jac})

        def c4(v, i=i):
            x, y, r = unpack(v)
            return float(1.0 - y[i] - r[i])
        def c4_jac(v, i=i):
            g = np.zeros_like(v)
            g[3 * i + 1] = -1.0
            g[3 * i + 2] = -1.0
            return g
        cons.append({'type': 'ineq', 'fun': c4, 'jac': c4_jac})

    for i in range(n):
        for j in range(i + 1, n):
            def cp(v, i=i, j=j):
                x, y, r = unpack(v)
                return float((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2 - (r[i] + r[j]) ** 2)
            def cp_jac(v, i=i, j=j):
                g = np.zeros_like(v)
                x, y, r = unpack(v)
                dx = float(x[i] - x[j])
                dy = float(y[i] - y[j])
                rij = float(r[i] + r[j])
                g[3 * i] = 2.0 * dx
                g[3 * j] = -2.0 * dx
                g[3 * i + 1] = 2.0 * dy
                g[3 * j + 1] = -2.0 * dy
                g[3 * i + 2] = -2.0 * rij
                g[3 * j + 2] = -2.0 * rij
                return g
            cons.append({'type': 'ineq', 'fun': cp, 'jac': cp_jac})

    def attempt(v_start, maxiter, ftol, current_circles):
        if time.time() - start > time_limit:
            return None
        res = minimize(
            obj,
            v_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            jac=obj_jac,
            options={
                'maxiter': max(50, maxiter),
                'ftol': ftol,
                'disp': False,
            },
        )
        if time.time() - start > time_limit:
            return None
        if res.x is None:
            return None
        v = res.x
        x, y, r = unpack(v)
        cand = [[float(x[i]), float(y[i]), float(r[i])] for i in range(n)]
        if _valid(cand, tol=1e-9) and sum(c[2] for c in cand) > sum(c[2] for c in current_circles) + 1e-12:
            return cand
        return None

    best = circles
    best_sum = sum(c[2] for c in circles)

    cand = attempt(v0, 50, 1e-14, circles)
    if cand is not None:
        best = cand
        best_sum = sum(c[2] for c in cand)
        cand2 = attempt(np.array([v for c in cand for v in c], dtype=float), 50, 1e-14, cand)
        if cand2 is not None and sum(c[2] for c in cand2) > best_sum + 1e-12:
            best = cand2
            best_sum = sum(c[2] for c in cand2)

    polished = _maximize_radii(best, tol=1e-10)
    if _valid(polished, tol=1e-9):
        s = sum(c[2] for c in polished)
        if s > best_sum + 1e-12:
            best = polished
            best_sum = s
            cand3 = attempt(np.array([v for c in polished for v in c], dtype=float), 50, 1e-14, polished)
            if cand3 is not None and sum(c[2] for c in cand3) > best_sum + 1e-12:
                best = cand3
                best_sum = sum(c[2] for c in cand3)

    return best


def solve(n, seed, budget):
    if n == 26:
        base = _base26()
        best = base
        best_sum = sum(c[2] for c in base)

        starts = [base]

        rng = random.Random(seed)
        starts.append(_perturb(base, rng, 0.004, 0.97, 1.03))
        rng = random.Random(seed + 1)
        starts.append(_perturb(base, rng, 0.006, 0.95, 1.05))
        rng = random.Random(seed + 4)
        starts.append(_rotate(base, rng.uniform(-0.2, 0.2)))
        starts.append(_rotate(base, math.pi / 6.0))
        starts.append(_rotate(base, math.pi / 12.0))

        rng = random.Random(seed + 12)
        starts.append(_perturb(_rotate(base, rng.uniform(0, math.pi / 2.0)), rng, 0.005, 0.95, 1.05))

        rng = random.Random(seed + 30)
        aggressive = _perturb(base, rng, 0.005, 0.9, 1.1)
        if _valid(aggressive, tol=1e-9):
            starts.append(aggressive)

        reflected = _reflect(base)
        if _valid(reflected, tol=1e-9):
            starts.append(reflected)

        start_time = time.time()
        for idx, cand in enumerate(starts):
            remaining = (budget if budget is not None else 15.0) - (idx * 2.0)
            if remaining <= 0:
                break
            improved = _local_improve(cand, time_limit=min(2.0, remaining), seed=seed)
            if _valid(improved, tol=1e-9):
                s = sum(c[2] for c in improved)
                if s > best_sum + 1e-12:
                    best = improved
                    best_sum = s

        total_budget = budget if budget is not None else 15.0
        remaining = total_budget - (time.time() - start_time)
        if remaining > 0.5:
            rng = random.Random(seed + 2)
            cand = _perturb(best, rng, 0.003, 0.98, 1.02)
            improved = _local_improve(cand, time_limit=min(0.8, remaining), seed=seed)
            if _valid(improved, tol=1e-9):
                s = sum(c[2] for c in improved)
                if s > best_sum + 1e-12:
                    best = improved
                    best_sum = s

            remaining = total_budget - (time.time() - start_time)
            if remaining > 0.5:
                rng = random.Random(seed + 3)
                cand = _perturb(best, rng, 0.002, 0.99, 1.01)
                improved = _local_improve(cand, time_limit=min(0.8, remaining), seed=seed)
                if _valid(improved, tol=1e-9):
                    s = sum(c[2] for c in improved)
                    if s > best_sum + 1e-12:
                        best = improved
                        best_sum = s

        restart_seed = seed + 20
        for _ in range(8):
            remaining = total_budget - (time.time() - start_time)
            if remaining <= 0.4:
                break
            rng = random.Random(restart_seed)
            restart_seed += 1
            cand = _perturb(best, rng, 0.0015, 0.98, 1.02)
            if not _valid(cand, tol=1e-9):
                continue
            improved = _local_improve(cand, time_limit=min(0.4, remaining), seed=seed)
            if _valid(improved, tol=1e-9):
                s = sum(c[2] for c in improved)
                if s > best_sum + 1e-12:
                    best = improved
                    best_sum = s

        rng = random.Random(seed + 50)
        for _ in range(2):
            remaining = total_budget - (time.time() - start_time)
            if remaining <= 0.3:
                break
            cand = _random_start(26, rng, 0.07)
            if cand is None or not _valid(cand, tol=1e-9):
                continue
            improved = _local_improve(cand, time_limit=min(0.8, remaining), seed=seed)
            if _valid(improved, tol=1e-9):
                s = sum(c[2] for c in improved)
                if s > best_sum + 1e-12:
                    best = improved
                    best_sum = s

        rng = random.Random(seed + 60)
        cand = _greedy_start(26, rng, 0.07)
        if cand is not None and _valid(cand, tol=1e-9):
            remaining = total_budget - (time.time() - start_time)
            if remaining > 0.2:
                improved = _local_improve(cand, time_limit=min(0.8, remaining), seed=seed)
                if _valid(improved, tol=1e-9):
                    s = sum(c[2] for c in improved)
                    if s > best_sum + 1e-12:
                        best = improved
                        best_sum = s

        final_polished = _local_improve(best, time_limit=0.5, seed=seed)
        if _valid(final_polished, tol=1e-9):
            s = sum(c[2] for c in final_polished)
            if s > best_sum + 1e-12:
                best = final_polished
                best_sum = s

        if not _valid(best, tol=1e-9):
            best = base
            best_sum = sum(c[2] for c in base)

        return best
    return _grid(n)
```
